## Here is a simple task from [freecodecamp](https://www.freecodecamp.org)

```javascript
const character = "#";
const count = 8;
const rows = [];

function padRow(rowNumber, rowCount) {
  return " ".repeat(rowCount - rowNumber) + character.repeat(2 * rowNumber - 1) + " ".repeat(rowCount - rowNumber);
}

let i = 1;
while (i <= count) {
  rows.push(padRow(i, count));
  i++;
}

let result = "";

for (const row of rows) {
  result += row + "\n";
}

console.log(result);
```
